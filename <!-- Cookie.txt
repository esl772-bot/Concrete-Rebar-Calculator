<!-- Cookie Consent Banner Component -->
<div id="cookieBanner" class="fixed bottom-4 left-4 right-4 md:left-auto md:max-w-sm bg-slate-950 border border-slate-800 text-white p-4 rounded-xl shadow-2xl z-50 hidden transition-all duration-300">
    <p class="text-xs text-slate-300 leading-relaxed">
        We utilize third-party cookies to optimize B2B banner matching and measure analytical job-site performance. View our <a href="privacy.html" class="underline text-blue-400">Privacy Policy</a>.
    </p>
    <div class="mt-3 flex justify-end gap-2 text-xs font-semibold">
        <button onclick="dismissBanner()" class="bg-slate-800 hover:bg-slate-700 px-3 py-1.5 rounded-md transition cursor-pointer">Decline</button>
        <button onclick="dismissBanner()" class="bg-blue-600 hover:bg-blue-500 px-3 py-1.5 rounded-md transition cursor-pointer">Accept</button>
    </div>
</div>

<script>
    // Simple state verification logic for tracking banner lifecycle
    function checkCookieConsent() {
        if (!localStorage.getItem("cookieConsentDismissed")) {
            document.getElementById("cookieBanner").classList.remove("hidden");
        }
    }
    function dismissBanner() {
        localStorage.setItem("cookieConsentDismissed", "true");
        document.getElementById("cookieBanner").classList.add("hidden");
    }
    window.addEventListener("DOMContentLoaded", checkCookieConsent);
</script>
